private static GachaEntry PickOne(
    IReadOnlyList<GachaEntry> pool, int totalWeight)
{
    // Random.Shared는 스레드 안전하다(.NET 6+).
    int roll = Random.Shared.Next(totalWeight);   // ← 주입 불가
    int cumulative = 0;
    foreach (var entry in pool)                   // ← O(n) 선형 스캔
    {
        cumulative += entry.Weight;
        if (roll < cumulative)
            return entry;
    }
    return pool[^1];
}

public T Pick(Random? random = null)
{
    var roll = (random ?? Random.Shared).Next(TotalWeight);

    // roll이 속한 누적 구간 = roll보다 큰 첫 원소의 위치.
    var found = Array.BinarySearch(_cumulative, roll);

}

{

    CurrentWorkSpeed = clamped;
    return true;
}
